Двухсторонний код для Google Таблицы (Списания + Сроки): 
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheets()[0];
  var rows = sheet.getDataRange().getDisplayValues();
  var result = [];
  for (var i = 1; i < rows.length; i++) {
    var r = rows[i];
    if (!r[0] && !r[3]) continue;
    result.push({
      id: String(r[0]),
      date: String(r[1]),
      time: String(r[2]),
      product: String(r[3]),
      qty: Number(String(r[4]).replace(",", ".")) || 0,
      unit: String(r[5]),
      reason: String(r[6]),
      bartender: String(r[7] || ""),
      comment: String(r[8] || "")
    });
  }
  result.reverse();

  var expSheet = ss.getSheetByName("Сроки");
  var expiries = [];
  if (expSheet) {
    var erows = expSheet.getDataRange().getDisplayValues();
    for (var j = 1; j < erows.length; j++) {
      var er = erows[j];
      if (!er[0] && !er[1]) continue;
      expiries.push({
        id: String(er[0]),
        product: String(er[1]),
        madeDate: String(er[2]),
        expiryDate: String(er[3]),
        volume: String(er[4] || ""),
        bartender: String(er[5] || "")
      });
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "ok", entries: result, expiries: expiries }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var data = JSON.parse(e.postData.contents);

  if (data.type === "expiry") {
    var expSheet = ss.getSheetByName("Сроки") || ss.insertSheet("Сроки");
    if (expSheet.getLastRow() === 0) {
      expSheet.appendRow(["ID", "Заготовка", "Дата открытия", "Годен до", "Объем", "Бармен"]);
    }
    if (data.action === "delete" && data.id) {
      var erows = expSheet.getDataRange().getDisplayValues();
      for (var k = erows.length - 1; k >= 1; k--) {
        if (String(erows[k][0]) === String(data.id)) {
          expSheet.deleteRow(k + 1);
          break;
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ status: "deleted_expiry" })).setMimeType(ContentService.MimeType.JSON);
    }
    expSheet.appendRow([data.id, data.product, "'" + data.madeDate, "'" + data.expiryDate, data.volume || "", data.bartender || ""]);
    return ContentService.createTextOutput(JSON.stringify({ status: "added_expiry" })).setMimeType(ContentService.MimeType.JSON);
  }

  var sheet = ss.getSheets()[0];
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["ID", "Дата", "Время", "Продукт", "Кол-во", "Ед.", "Причина", "Бармен", "Комментарий"]);
  }
  if (data.action === "delete" && data.id) {
    var rows = sheet.getDataRange().getDisplayValues();
    for (var i = rows.length - 1; i >= 1; i--) {
      if (String(rows[i][0]) === String(data.id)) {
        sheet.deleteRow(i + 1);
        break;
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ status: "deleted" })).setMimeType(ContentService.MimeType.JSON);
  }
  sheet.appendRow([
    data.id,
    "'" + data.date,
    "'" + (data.time || ""),
    data.product,
    data.qty,
    data.unit,
    data.reason,
    data.bartender || "",
    data.comment || ""
  ]);
  return ContentService.createTextOutput(JSON.stringify({ status: "added" })).setMimeType(ContentService.MimeType.JSON);
}

# Готовый Telegram-бот для списаний на баре (aiogram 3.x)
import asyncio, csv
from datetime import datetime
from aiogram import Bot, Dispatcher, F
from aiogram.types import Message, ReplyKeyboardMarkup, KeyboardButton
from aiogram.fsm.context import FSMContext
from aiogram.fsm.state import State, StatesGroup

TOKEN = "ВСТАВЬ_ТОКЕН_ОТ_BOTFATHER"

REASONS = [
    [KeyboardButton(text="🥀 Порча продукта"), KeyboardButton(text="🎁 Комплимент гостю")],
    [KeyboardButton(text="🧪 Проработка"), KeyboardButton(text="👔 Выдача начальству")],
    [KeyboardButton(text="💥 Бой / Разлив")]
]

class WriteOff(StatesGroup):
    product = State()
    qty = State()
    reason = State()

dp = Dispatcher()

@dp.message(F.text == "/start")
async def start(msg: Message, state: FSMContext):
    await state.set_state(WriteOff.product)
    await msg.answer("🍸 Введи название продукта для списания:")

@dp.message(WriteOff.product)
async def get_product(msg: Message, state: FSMContext):
    await state.update_data(product=msg.text)
    await state.set_state(WriteOff.qty)
    await msg.answer("⚖️ Укажи количество и единицу (например: 200 г, 50 мл, 2 шт):")

@dp.message(WriteOff.qty)
async def get_qty(msg: Message, state: FSMContext):
    await state.update_data(qty=msg.text)
    await state.set_state(WriteOff.reason)
    kb = ReplyKeyboardMarkup(keyboard=REASONS, resize_keyboard=True, one_time_keyboard=True)
    await msg.answer("📌 Выбери причину списания:", reply_markup=kb)

@dp.message(WriteOff.reason)
async def get_reason(msg: Message, state: FSMContext):
    data = await state.get_data()
    now = datetime.now().strftime("%Y-%m-%d %H:%M")
    user = msg.from_user.first_name if msg.from_user else "Бармен"
    row = [now, data["product"], data["qty"], msg.text, user]
    with open("writeoffs.csv", "a", newline="", encoding="utf-8") as f:
        csv.writer(f).writerow(row)
    await state.clear()
    await msg.answer(f"✅ Списано ({user}):\n📅 {now}\n🍋 {data['product']} — {data['qty']}\n📌 {msg.text}")

async def main():
    bot = Bot(TOKEN)
    await dp.start_polling(bot)

if __name__ == "__main__":
    asyncio.run(main())
