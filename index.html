Ссылка на скрипт VagonBar вшита в код по умолчанию для всех устройств.

function authorizeDrive(){
  var folders=DriveApp.getFoldersByName('VagonBar_Photos');
  var folder=folders.hasNext()?folders.next():DriveApp.createFolder('VagonBar_Photos');
  var ss=SpreadsheetApp.getActiveSpreadsheet();
  ['Списания','Сроки','ТК'].forEach(function(n){if(!ss.getSheetByName(n))ss.insertSheet(n);});
  Logger.log('Готово! '+folder.getUrl());
}
function doGet(e){return handleReq(e);}
function doPost(e){return handleReq(e);}
function saveImageToDrive(base64Str,name){
  try{
    if(!base64Str||base64Str.indexOf('data:image')!==0)return base64Str||'';
    var folders=DriveApp.getFoldersByName('VagonBar_Photos');
    var folder=folders.hasNext()?folders.next():DriveApp.createFolder('VagonBar_Photos');
    var split=base64Str.split(',');
    var mime=split[0].match(/:(.*?);/)[1]||'image/jpeg';
    var ext=mime.indexOf('webp')>-1?'webp':(mime.indexOf('png')>-1?'png':'jpg');
    var bytes=Utilities.base64Decode(split[1]);
    var blob=Utilities.newBlob(bytes,mime,(name||'tk')+'_'+new Date().getTime()+'.'+ext);
    var file=folder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK,DriveApp.Permission.VIEW);
    return 'https://drive.google.com/thumbnail?id='+file.getId()+'&sz=w1000';
  }catch(err){return '';}
}
function handleReq(e){
  var lock=LockService.getScriptLock();lock.tryLock(10000);
  try{
    var ss=SpreadsheetApp.getActiveSpreadsheet();
    var sW=ss.getSheetByName('Списания')||ss.insertSheet('Списания');
    var sE=ss.getSheetByName('Сроки')||ss.insertSheet('Сроки');
    var sT=ss.getSheetByName('ТК')||ss.insertSheet('ТК');
    if(sW.getLastRow()===0)sW.appendRow(['ID','Дата','Время','Бармен','Позиция','Кол-во','Ед.','Причина','Комментарий']);
    if(sE.getLastRow()===0)sE.appendRow(['ID','Позиция','Дата открытия','Срок (дней)','Годен до','Объем','Комментарий']);
    if(sT.getLastRow()===0)sT.appendRow(['ID','Название','Категория','Выход','Ед. выхода','Состав','Метод / Подача','JSON']);
    if(e&&e.postData&&e.postData.contents){
      var d=JSON.parse(e.postData.contents);
      if(d.action==='delete'){
        if(d.id==='__CLEAR_WRITEOFFS__'){if(sW.getLastRow()>1)sW.deleteRows(2,sW.getLastRow()-1);}
        else{[sW,sE,sT].forEach(function(sh){var r=sh.getDataRange().getValues();for(var i=r.length-1;i>=1;i--){if(String(r[i][0])===String(d.id))sh.deleteRow(i+1);}});}
        return ContentService.createTextOutput(JSON.stringify({status:'deleted'})).setMimeType(ContentService.MimeType.JSON);
      }
      if(d.type==='expiry'){
        sE.appendRow([d.id,d.name,d.startDate,d.days,d.endDate,d.volume||'',d.note||'']);
      }else if(d.type==='tk'){
        var rows=sT.getDataRange().getValues(),found=false;
        var noteText=d.note||d.method||'';
        var rawImg=d.imageDrive||d.image||'';
        var finalImg=rawImg.indexOf('data:image')===0?saveImageToDrive(rawImg,d.name):rawImg;
        var ings=Array.isArray(d.ingredients)?d.ingredients:[];
        if(ings.length>0){ings[0]._note=noteText;if(finalImg)ings[0]._image=finalImg;else if(d.image&&!finalImg)ings[0]._image=d.image;}
        var ingText=ings.map(function(x){return x.name+': '+x.qty+' '+x.unit;}).join(', ');
        var rowData=[d.id,d.name,d.category,d.yieldQty,d.yieldUnit,ingText,noteText,JSON.stringify(ings)];
        for(var i=1;i<rows.length;i++){if(String(rows[i][0])===String(d.id)){sT.getRange(i+1,1,1,8).setValues([rowData]);found=true;break;}}
        if(!found)sT.appendRow(rowData);
        return ContentService.createTextOutput(JSON.stringify({status:'ok',imageUrl:finalImg||d.image||''})).setMimeType(ContentService.MimeType.JSON);
      }else{
        sW.appendRow([d.id,d.date,d.time,d.staff,d.item,d.qty,d.unit,d.reason,d.comment||'']);
      }
      return ContentService.createTextOutput(JSON.stringify({status:'ok'})).setMimeType(ContentService.MimeType.JSON);
    }
    var wRows=sW.getDataRange().getValues().slice(1).map(function(r){return{id:String(r[0]),date:r[1],time:r[2],staff:r[3],item:r[4],qty:Number(r[5]),unit:r[6],reason:r[7],comment:r[8]};});
    var eRows=sE.getDataRange().getValues().slice(1).map(function(r){return{id:String(r[0]),name:r[1],startDate:r[2],days:Number(r[3]),endDate:r[4],volume:r[5],note:r[6]};});
    var tRows=sT.getDataRange().getValues().slice(1).map(function(r){var ig=[];try{ig=JSON.parse(r[7]||'[]');}catch(err){}var nt=r[6]||(ig[0]&&ig[0]._note)||'';var im=(ig[0]&&ig[0]._image)||'';return{id:String(r[0]),name:r[1],category:r[2],yieldQty:Number(r[3]),yieldUnit:r[4],note:nt,image:im,ingredients:ig};});
    return ContentService.createTextOutput(JSON.stringify({writeoffs:wRows,expiries:eRows,recipes:tRows})).setMimeType(ContentService.MimeType.JSON);
  }finally{lock.releaseLock();}
}
