Общий журнал для всех сменщиков (2-сторонняя синхронизация)

Твоя ссылка на Google Таблицу вшита по умолчанию! Списания и ⏳ Сроки годности ведутся на первом листе в разных столбиках, а для 📖 ТК (Техкарт) скрипт автоматически ведёт отдельную вкладку «ТК» внизу твоей Google Таблицы.

Двухсторонний код для Google Таблицы (с отдельной вкладкой «ТК»): 
// Лист 1: Списания слева (колонки A–I), Сроки хранения справа (колонки K–P).
// Отдельная вкладка «ТК» (создаётся автоматически): хранит базу техкарт бара!
function getMainSheet() {
  return SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
}

function getOrCreateTkSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var tkSheet = ss.getSheetByName("ТК");
  if (!tkSheet) {
    tkSheet = ss.insertSheet("ТК");
    tkSheet.getRange(1, 1, 1, 8).setValues([[
      "ID ТК", "Название", "Категория", "Выход", "Ед. выхода", "Состав (ингредиенты)", "Метод / Подача", "JSON"
    ]]);
  } else if (tkSheet.getLastRow() === 0) {
    tkSheet.getRange(1, 1, 1, 8).setValues([[
      "ID ТК", "Название", "Категория", "Выход", "Ед. выхода", "Состав (ингредиенты)", "Метод / Подача", "JSON"
    ]]);
  }
  return tkSheet;
}

function ensureHeadersAndMigrate(sheet) {
  if (sheet.getLastRow() === 0) {
    sheet.getRange(1, 1, 1, 16).setValues([[
      "ID", "Дата", "Время", "Продукт", "Кол-во", "Ед.", "Причина", "Бармен", "Комментарий",
      "",
      "ID срока", "Заготовка / Продукт", "Объем / Тара", "Дата откр.", "Годен до", "Бармен"
    ]]);
    return;
  }
  if (!sheet.getRange(1, 11).getValue()) {
    sheet.getRange(1, 11, 1, 6).setValues([[
      "ID срока", "Заготовка / Продукт", "Объем / Тара", "Дата откр.", "Годен до", "Бармен"
    ]]);
  }
}

function doGet(e) {
  var sheet = getMainSheet();
  var tkSheet = getOrCreateTkSheet();
  ensureHeadersAndMigrate(sheet);

  var rows = sheet.getDataRange().getDisplayValues();
  var entries = [];
  var expiries = [];
  var recipes = [];

  for (var i = 1; i < rows.length; i++) {
    var r = rows[i];
    if ((r[0] || r[3]) && String(r[6]) !== "__EXPIRY__" && String(r[6]) !== "__RECIPE__") {
      entries.push({
        id: String(r[0]),
        date: String(r[1]),
        time: String(r[2]),
        product: String(r[3]),
        qty: Number(String(r[4]).replace(",", ".")) || 0,
        unit: String(r[5]),
        reason: String(r[6]),
        bartender: String(r[7] || ""),
        comment: String(r[8] || "")
      });
    }
    if (r[10] || r[11]) {
      expiries.push({
        id: String(r[10]),
        product: String(r[11]),
        volume: String(r[12] || ""),
        madeDate: String(r[13] || ""),
        expiryDate: String(r[14] || ""),
        bartender: String(r[15] || "")
      });
    }
  }

  var tkRows = tkSheet.getDataRange().getDisplayValues();
  for (var j = 1; j < tkRows.length; j++) {
    var tr = tkRows[j];
    if (tr[0] || tr[1]) {
      var parsedIngs = [];
      if (tr[7]) {
        try {
          var parsedJson = JSON.parse(tr[7]);
          if (Array.isArray(parsedJson)) parsedIngs = parsedJson;
          else if (parsedJson && Array.isArray(parsedJson.ingredients)) parsedIngs = parsedJson.ingredients;
        } catch (err) {}
      }
      recipes.push({
        id: String(tr[0]),
        name: String(tr[1]),
        category: String(tr[2] || "Коктейли"),
        yieldQty: Number(String(tr[3]).replace(",", ".")) || 1,
        yieldUnit: String(tr[4] || "порц"),
        ingredientsText: String(tr[5] || ""),
        method: String(tr[6] || ""),
        note: String(tr[6] || ""),
        ingredients: parsedIngs
      });
    }
  }

  entries.reverse();
  expiries.reverse();
  return ContentService.createTextOutput(JSON.stringify({
    status: "ok",
    entries: entries,
    expiries: expiries,
    recipes: recipes
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = getMainSheet();
  var tkSheet = getOrCreateTkSheet();
  ensureHeadersAndMigrate(sheet);
  var data = JSON.parse(e.postData.contents);
  var rows = sheet.getDataRange().getDisplayValues();

  // Полная очистка журнала списаний (колонки A-I)
  if (data.action === "clear_writeoffs" || (data.action === "delete" && data.id === "__CLEAR_WRITEOFFS__")) {
    for (var i = rows.length - 1; i >= 1; i--) {
      if (rows[i][0] || rows[i][3]) {
        sheet.getRange(i + 1, 1, 1, 9).clearContent();
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ status: "cleared" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // Удаление записи (списание, срок или ТК)
  if (data.action === "delete" && data.id) {
    for (var i = rows.length - 1; i >= 1; i--) {
      if (String(rows[i][0]) === String(data.id)) {
        sheet.getRange(i + 1, 1, 1, 9).clearContent();
        break;
      }
      if (String(rows[i][10]) === String(data.id)) {
        sheet.getRange(i + 1, 11, 1, 6).clearContent();
        break;
      }
    }
    var tkRows = tkSheet.getDataRange().getDisplayValues();
    for (var k = tkRows.length - 1; k >= 1; k--) {
      if (String(tkRows[k][0]) === String(data.id)) {
        tkSheet.deleteRow(k + 1);
        break;
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ status: "deleted" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // Сохранение или обновление Техкарты на отдельной вкладке «ТК»
  if (data.type === "recipe" || data.reason === "__RECIPE__") {
    var rec = data.recipe || data;
    var ings = Array.isArray(data.ingredients) ? data.ingredients : (rec.ingredients || []);
    var ingSummary = ings.map(function(ing) {
      return ing.name + " — " + ing.qty + " " + ing.unit;
    }).join(" | ");
    var methodStr = data.method || data.note || rec.note || rec.method || "";
    var tkData = tkSheet.getDataRange().getDisplayValues();
    var targetTkRow = -1;
    for (var m = 1; m < tkData.length; m++) {
      if (String(tkData[m][0]) === String(data.id)) {
        targetTkRow = m + 1;
        break;
      }
    }
    if (targetTkRow === -1) targetTkRow = tkSheet.getLastRow() + 1;
    tkSheet.getRange(targetTkRow, 1, 1, 8).setValues([[
      data.id,
      data.name || rec.name || data.product || "",
      data.category || rec.category || "Коктейли",
      data.yieldQty || rec.yieldQty || 1,
      data.yieldUnit || rec.yieldUnit || "порц",
      ingSummary,
      methodStr,
      JSON.stringify(ings)
    ]]);
    return ContentService.createTextOutput(JSON.stringify({ status: "recipe_saved" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // Запись срока хранения в правые столбики K–P
  if (data.type === "expiry" || data.reason === "__EXPIRY__") {
    var targetRow = 2;
    for (var r = 1; r < rows.length; r++) {
      if (rows[r][10] || rows[r][11]) targetRow = r + 2;
    }
    sheet.getRange(targetRow, 11, 1, 6).setValues([[
      data.id,
      data.product,
      data.volume || data.unit || "",
      "'" + (data.madeDate || data.date || ""),
      "'" + (data.expiryDate || data.comment || ""),
      data.bartender || ""
    ]]);
    return ContentService.createTextOutput(JSON.stringify({ status: "expiry_added" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // Запись обычного списания в левые столбики A–I
  var writeRow = 2;
  for (var w = 1; w < rows.length; w++) {
    if (rows[w][0] || rows[w][3]) writeRow = w + 2;
  }
  sheet.getRange(writeRow, 1, 1, 9).setValues([[
    data.id,
    "'" + data.date,
    "'" + (data.time || ""),
    data.product,
    data.qty,
    data.unit,
    data.reason,
    data.bartender || "",
    data.comment || ""
  ]]);
  return ContentService.createTextOutput(JSON.stringify({ status: "added" }))
    .setMimeType(ContentService.MimeType.JSON);
}

Код отдельного Telegram-бота (по желанию)

# Готовый Telegram-бот для списаний на баре (aiogram 3.x)
import asyncio, csv
from datetime import datetime
from aiogram import Bot, Dispatcher, F
from aiogram.types import Message, ReplyKeyboardMarkup, KeyboardButton
from aiogram.fsm.context import FSMContext
from aiogram.fsm.state import State, StatesGroup

TOKEN = "ВСТАВЬ_ТОКЕН_ОТ_BOTFATHER"

REASONS = [
    [KeyboardButton(text="🥀 Порча продукта"), KeyboardButton(text="🎁 Комплимент гостю")],
    [KeyboardButton(text="🧪 Проработка"), KeyboardButton(text="👔 Выдача начальству")],
    [KeyboardButton(text="💥 Бой / Разлив")]
]

class WriteOff(StatesGroup):
    product = State()
    qty = State()
    reason = State()

dp = Dispatcher()

@dp.message(F.text == "/start")
async def start(msg: Message, state: FSMContext):
    await state.set_state(WriteOff.product)
    await msg.answer("🍸 Введи название продукта для списания:")

@dp.message(WriteOff.product)
async def get_product(msg: Message, state: FSMContext):
    await state.update_data(product=msg.text)
    await state.set_state(WriteOff.qty)
    await msg.answer("⚖️ Укажи количество и единицу (например: 200 г, 50 мл, 2 шт):")

@dp.message(WriteOff.qty)
async def get_qty(msg: Message, state: FSMContext):
    await state.update_data(qty=msg.text)
    await state.set_state(WriteOff.reason)
    kb = ReplyKeyboardMarkup(keyboard=REASONS, resize_keyboard=True, one_time_keyboard=True)
    await msg.answer("📌 Выбери причину списания:", reply_markup=kb)

@dp.message(WriteOff.reason)
async def get_reason(msg: Message, state: FSMContext):
    data = await state.get_data()
    now = datetime.now().strftime("%Y-%m-%d %H:%M")
    user = msg.from_user.first_name if msg.from_user else "Бармен"
    row = [now, data["product"], data["qty"], msg.text, user]
    with open("writeoffs.csv", "a", newline="", encoding="utf-8") as f:
        csv.writer(f).writerow(row)
    await state.clear()
    await msg.answer(f"✅ Списано ({user}):\n📅 {now}\n🍋 {data['product']} — {data['qty']}\n📌 {msg.text}")

async def main():
    bot = Bot(TOKEN)
    await dp.start_polling(bot)

if __name__ == "__main__":
    asyncio.run(main())
